import sqlite3

# Q3 abandonment spike: root cause
con = sqlite3.connect("transactions.db")

q = """
  SELECT quarter,
    ROUND(100.0 * SUM(abandoned)
      / COUNT(*), 1) AS rate
  FROM transactions
  GROUP BY quarter
"""

df = pd.read_sql(q, con)
